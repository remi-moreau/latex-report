% -----------------
% FLUSH TWO COLUMNS
% -----------------
% \flushtwocol{arg1}{arg2, arg3, arg4, ...}
% - arg1 : texte courant, placé normalement
% - arg2, arg3, ... : paires formant un bloc à deux colonnes (mode texte ;
%   mettre des $...$ soi-même pour les maths). Un nombre impair est accepté :
%   le dernier élément est placé seul en colonne gauche.
% - Si largeur(arg1) + largeur(bloc) <= \linewidth : le bloc suit arg1, avec
%   arg1 en face de la première ligne du bloc (en face de arg2).
% - Sinon : le bloc est mis à la ligne, aligné à gauche avec une indentation
%   d'un \quad, sans interligne supplémentaire (simple saut de ligne, même
%   \baselineskip).
% - Si le bloc seul dépasse \linewidth, il déborde (overfull hbox).
\ExplSyntaxOn

\tl_new:N \l_flushtwocol_body_tl
\int_new:N \l_flushtwocol_count_int
\box_new:N \l_flushtwocol_arg_box
\box_new:N \l_flushtwocol_block_box

\NewDocumentCommand{\flushtwocol}{ m m }
  {
    % 1. Construction du corps du bloc : lignes "gauche &\ droite"
    \tl_clear:N \l_flushtwocol_body_tl
    \int_zero:N \l_flushtwocol_count_int

    \clist_map_inline:nn { #2 }
      {
        \int_incr:N \l_flushtwocol_count_int

        \int_if_even:nTF { \l_flushtwocol_count_int }
          {
            % CAS PAIR (colonne de droite) : on ajoute juste l'argument
            \tl_put_right:Nn \l_flushtwocol_body_tl { ##1 }
          }
          {
            % CAS IMPAIR (colonne de gauche) :
            % S'il ne s'agit pas du tout premier argument, on ajoute un saut de ligne \\
            \int_compare:nNnF { \l_flushtwocol_count_int } = { 1 }
              { \tl_put_right:Nn \l_flushtwocol_body_tl { \\ } }

            % On ajoute l'argument, suivi de l'esperluette et de l'espace demandés
            \tl_put_right:Nn \l_flushtwocol_body_tl { ##1 &\  }
          }
      }

    % 2. Mesure de arg1 et du bloc (tabular[t] : arg1 reste en face de arg2 ;
    %    colonne 1 alignée à droite, colonne 2 à gauche, comme un align)
    \hbox_set:Nn \l_flushtwocol_arg_box { #1 }
    \hbox_set:Nn \l_flushtwocol_block_box
      {
        \begin{tabular}[t]{@{}rl@{}}
          \l_flushtwocol_body_tl
        \end{tabular}
      }

    % 3. Bascule automatique : en ligne si ça tient, sinon à la ligne,
    %    aligné à gauche avec une indentation d'un \quad (1em)
    \dim_compare:nTF
      { \box_wd:N \l_flushtwocol_arg_box + \box_wd:N \l_flushtwocol_block_box > \linewidth }
      {
        #1\\*\hspace*{2em}\box_use:N \l_flushtwocol_block_box
      }
      {
        #1~\box_use:N \l_flushtwocol_block_box
      }
  }

\ExplSyntaxOff

% Variable de liste interne
\clist_new:N \l_acc_clist

% Macro \leftacc { arg1, arg2, ... }
\NewDocumentCommand{\leftacc}{ m }
  {
    \clist_set:Nn \l_acc_clist { #1 }
    \ensuremath
      {
        \left\{
        \begin{array}{l}
          \clist_use:Nn \l_acc_clist { \\ }
        \end{array}
        \right.
      }
  }

% Macro \rightacc { arg1, arg2, ... }
\NewDocumentCommand{\rightacc}{ m }
  {
    \clist_set:Nn \l_acc_clist { #1 }
    \ensuremath
      {
        \left.
        \begin{array}{r}
          \clist_use:Nn \l_acc_clist { \\ }
        \end{array}
        \right\}
      }
  }

  backgroundcolor=none,      % transparent background
  linecolor=black,           % line color
  linewidth=3pt,             % line thickness
  leftline=true,             % line on the left only
  topline=false,             % no top line
  bottomline=false,          % no bottom line
  rightline=false,           % no right line
  innertopmargin=0pt,        % inner margins
  innerbottommargin=10pt,
  innerleftmargin=10pt,
  innerrightmargin=10pt
}

  \begin{mdframed}[style=dembox]
